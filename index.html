function initMatchingGame() { if (!cards || cards.length === 0) return; const grid = document.getElementById('matching-grid'); grid.innerHTML = ''; // 1. Filter out any cards that are missing german or english text const validCards = cards.filter(item => item && item.german && item.english && item.german.trim() !== "" && item.english.trim() !== ""); if (validCards.length < 4) { grid.innerHTML = '
Not enough valid translation pairs found in JSON.
'; return; } // 2. Shuffle and pick 4 valid items let shuffled = [...validCards].sort(() => 0.5 - Math.random()); let roundCards = shuffled.slice(0, 4); let gameTiles = []; roundCards.forEach((item, index) => { // Push explicit pairs with unique IDs gameTiles.push({ text: item.german.trim(), pairId: index, type: 'german' }); gameTiles.push({ text: item.english.trim(), pairId: index, type: 'english' }); }); // 3. Shuffle the tiles for the board gameTiles.sort(() => 0.5 - Math.random()); gameTiles.forEach(tile => { const div = document.createElement('div'); div.className = 'match-card'; div.innerText = tile.text; div.dataset.pairId = tile.pairId; div.dataset.type = tile.type; div.onclick = () => selectTile(div); grid.appendChild(div); }); }
